# 2.7 Python位运算

# Python位运算详细总结

## 1. 位运算符概览

### 位运算符表格

| 运算符 | 名称 | 描述 | 示例 | 优先级 |
| --- | --- | --- | --- | --- |
| `&` | 按位与 | 对应位都为1时结果为1 | `5 & 3 = 1` | 9 |
| ` | ` | 按位或 | 对应位有一个为1时结果为1 | `5 |
| `^` | 按位异或 | 对应位不同时结果为1 | `5 ^ 3 = 6` | 10 |
| `~` | 按位取反 | 0变1，1变0 | `~5 = -6` | 5 |
| `<<` | 左移 | 各位向左移动，右边补0 | `5 << 1 = 10` | 8 |
| `>>` | 右移 | 各位向右移动，左边补符号位 | `5 >> 1 = 2` | 8 |

## 2. 位运算符详细说明

### 2.1 按位与 (&)

**基本原理**：对应位都为1时结果为1，否则为0

```python
# 示例：5 & 3
# 5的二进制: 0101
# 3的二进制: 0011
# 结果:      0001 (1)

a = 5    # 0101
b = 3    # 0011
result = a & b  # 0001 = 1
print(f"{a} & {b} = {result}")  # 5 & 3 = 1

# 更多示例
print(12 & 7)   # 1100 & 0111 = 0100 = 4
print(15 & 15)  # 1111 & 1111 = 1111 = 15
print(8 & 0)    # 1000 & 0000 = 0000 = 0
```

**实际应用**：

```python
# 1. 检查奇偶性
def is_odd(n):
    return n & 1 == 1

print(is_odd(5))   # True
print(is_odd(4))   # False

# 2. 清除特定位（置0）
# 清除第3位（从0开始计数）
num = 13  # 1101
mask = ~(1 << 2)  # 11111111111111111111111111111011
result = num & mask  # 1001 = 9
print(f"清除第3位后: {result}")

# 3. 提取特定位
def get_bit(num, position):
    return (num >> position) & 1

num = 13  # 1101
print(f"第0位: {get_bit(num, 0)}")  # 1
print(f"第1位: {get_bit(num, 1)}")  # 0
print(f"第2位: {get_bit(num, 2)}")  # 1
print(f"第3位: {get_bit(num, 3)}")  # 1
```

### 2.2 按位或 (|)

**基本原理**：对应位有一个为1时结果为1，否则为0

```python
# 示例：5 | 3
# 5的二进制: 0101
# 3的二进制: 0011
# 结果:      0111 (7)

a = 5    # 0101
b = 3    # 0011
result = a | b  # 0111 = 7
print(f"{a} | {b} = {result}")  # 5 | 3 = 7

# 更多示例
print(12 | 7)   # 1100 | 0111 = 1111 = 15
print(8 | 0)    # 1000 | 0000 = 1000 = 8
print(0 | 0)    # 0000 | 0000 = 0000 = 0
```

**实际应用**：

```python
# 1. 设置特定位（置1）
def set_bit(num, position):
    return num | (1 << position)

num = 8  # 1000
result = set_bit(num, 1)  # 1010 = 10
print(f"设置第1位后: {result}")

# 2. 合并权限标志
class Permissions:
    READ = 1    # 0001
    WRITE = 2   # 0010
    EXECUTE = 4 # 0100
    DELETE = 8  # 1000

# 用户拥有读和写权限
user_perms = Permissions.READ | Permissions.WRITE
print(f"用户权限: {user_perms}")  # 3

# 检查权限
def has_permission(user_perms, permission):
    return (user_perms & permission) != 0

print(f"有读权限: {has_permission(user_perms, Permissions.READ)}")     # True
print(f"有写权限: {has_permission(user_perms, Permissions.WRITE)}")    # True
print(f"有执行权限: {has_permission(user_perms, Permissions.EXECUTE)}") # False
```

### 2.3 按位异或 (^)

**基本原理**：对应位不同时结果为1，相同时为0

```python
# 示例：5 ^ 3
# 5的二进制: 0101
# 3的二进制: 0011
# 结果:      0110 (6)

a = 5    # 0101
b = 3    # 0011
result = a ^ b  # 0110 = 6
print(f"{a} ^ {b} = {result}")  # 5 ^ 3 = 6

# 更多示例
print(12 ^ 7)   # 1100 ^ 0111 = 1011 = 11
print(15 ^ 15)  # 1111 ^ 1111 = 0000 = 0
print(8 ^ 0)    # 1000 ^ 0000 = 1000 = 8
```

**实际应用**：

```python
# 1. 翻转特定位
def toggle_bit(num, position):
    return num ^ (1 << position)

num = 13  # 1101
result = toggle_bit(num, 1)  # 1111 = 15
print(f"翻转第1位后: {result}")

# 2. 不使用临时变量交换两个数
def swap_numbers(a, b):
    a = a ^ b
    b = a ^ b  # (a ^ b) ^ b = a
    a = a ^ b  # (a ^ b) ^ a = b
    return a, b

x, y = 10, 20
print(f"交换前: x={x}, y={y}")
x, y = swap_numbers(x, y)
print(f"交换后: x={x}, y={y}")

# 3. 简单加密解密
def simple_encrypt(text, key):
    return ''.join(chr(ord(c) ^ key) for c in text)

def simple_decrypt(encrypted_text, key):
    return simple_encrypt(encrypted_text, key)  # 异或两次还原

message = "Hello"
key = 42
encrypted = simple_encrypt(message, key)
decrypted = simple_decrypt(encrypted, key)

print(f"原文: {message}")
print(f"加密: {encrypted}")
print(f"解密: {decrypted}")

# 4. 找到只出现一次的数字
def find_single_number(numbers):
    result = 0
    for num in numbers:
        result ^= num
    return result

# 其他数字都出现两次，只有一个出现一次
nums = [4, 2, 1, 2, 4]
print(f"只出现一次的数字: {find_single_number(nums)}")  # 1
```

### 2.4 按位取反 (~)

**基本原理**：0变1，1变0（注意Python使用补码表示负数）

```python
# 示例：~5
# 5的二进制:  00000000000000000000000000000101
# 取反后:    11111111111111111111111111111010
# 这是-6的补码表示

a = 5
result = ~a
print(f"~{a} = {result}")  # ~5 = -6

# 数学关系：~x = -(x + 1)
print(f"~5 = -(5 + 1) = {-6}")

# 更多示例
print(~0)   # -1
print(~1)   # -2
print(~-1)  # 0
print(~-2)  # 1
```

**实际应用**：

```python
# 1. 获取相反数（减1后取反）
def negate(x):
    return ~x + 1

print(f"5的相反数: {negate(5)}")   # -5
print(f"-3的相反数: {negate(-3)}") # 3

# 2. 检查是否为2的幂次方
def is_power_of_two(n):
    return n > 0 and (n & (n - 1)) == 0

print(f"8是2的幂次方: {is_power_of_two(8)}")    # True
print(f"10是2的幂次方: {is_power_of_two(10)}")  # False

# 3. 快速计算绝对值（针对32位整数）
def abs_32bit(x):
    mask = x >> 31  # 如果x为负数，mask为-1（全1），否则为0
    return (x + mask) ^ mask

print(f"abs(5): {abs_32bit(5)}")    # 5
print(f"abs(-5): {abs_32bit(-5)}")  # 5
```

### 2.5 左移 (<<)

**基本原理**：各二进制位向左移动指定位数，右边补0

```python
# 示例：5 << 1
# 5的二进制: 0101
# 左移1位:  1010 (10)

a = 5    # 0101
result = a << 1  # 1010 = 10
print(f"{a} << 1 = {result}")  # 5 << 1 = 10

# 更多示例
print(5 << 2)   # 0101 -> 10100 = 20
print(1 << 3)   # 0001 -> 1000 = 8
print(0 << 5)   # 0000 -> 00000 = 0

# 数学关系：x << n = x * 2^n
print(f"5 * 2^3 = {5 * 8} = {5 << 3}")  # 40
```

**实际应用**：

```python
# 1. 快速乘以2的幂次方
def fast_multiply_by_power_of_two(x, n):
    return x << n

print(f"7 * 8 = {fast_multiply_by_power_of_two(7, 3)}")  # 56

# 2. 生成标志位
flags = 0
flags |= 1 << 0  # 设置第0位
flags |= 1 << 2  # 设置第2位
flags |= 1 << 4  # 设置第4位
print(f"标志位: {bin(flags)}")  # 0b10101

# 3. RGB颜色值组合
def rgb_to_int(r, g, b):
    return (r << 16) | (g << 8) | b

red = 255
green = 128
blue = 64
color = rgb_to_int(red, green, blue)
print(f"RGB({red}, {green}, {blue}) = {color} (0x{color:06x})")

# 从整数中提取RGB值
def int_to_rgb(color):
    r = (color >> 16) & 0xFF
    g = (color >> 8) & 0xFF
    b = color & 0xFF
    return r, g, b

r, g, b = int_to_rgb(color)
print(f"从{color}提取RGB: ({r}, {g}, {b})")
```

### 2.6 右移 (>>)

**基本原理**：各二进制位向右移动指定位数，左边补符号位（正数补0，负数补1）

```python
# 示例：5 >> 1
# 5的二进制: 0101
# 右移1位:  0010 (2)

a = 5    # 0101
result = a >> 1  # 0010 = 2
print(f"{a} >> 1 = {result}")  # 5 >> 1 = 2

# 更多示例
print(16 >> 2)  # 10000 -> 00100 = 4
print(8 >> 3)   # 1000 -> 0001 = 1
print(0 >> 5)   # 0000 -> 00000 = 0

# 负数右移（算术右移）
print(-5 >> 1)  # -3 (保持符号位)

# 数学关系：x >> n = x // 2^n (对于正数)
print(f"15 // 4 = {15 // 4} = {15 >> 2}")  # 3
```

**实际应用**：

```python
# 1. 快速除以2的幂次方
def fast_divide_by_power_of_two(x, n):
    return x >> n

print(f"56 // 8 = {fast_divide_by_power_of_two(56, 3)}")  # 7

# 2. 二进制转十进制
def binary_to_decimal(binary_str):
    decimal = 0
    for i, bit in enumerate(reversed(binary_str)):
        if bit == '1':
            decimal |= 1 << i
    return decimal

print(f"1010转十进制: {binary_to_decimal('1010')}")  # 10

# 3. 获取整数的二进制表示
def get_binary_representation(n, bits=32):
    return [(n >> i) & 1 for i in range(bits-1, -1, -1)]

num = 13
binary = get_binary_representation(num, 8)
print(f"{num}的二进制: {''.join(map(str, binary))}")  # 00001101

# 4. 计算汉明重量（1的个数）
def hamming_weight(n):
    count = 0
    while n:
        count += n & 1
        n >>= 1
    return count

print(f"13(1101)的汉明重量: {hamming_weight(13)}")  # 3
```

## 3. 位运算的实际应用场景

### 3.1 权限管理系统

```python
class PermissionSystem:
    READ = 1 << 0    # 1
    WRITE = 1 << 1   # 2
    EXECUTE = 1 << 2 # 4
    DELETE = 1 << 3  # 8
    ADMIN = 1 << 4   # 16
    
    def __init__(self):
        self.user_permissions = {}
    
    def grant_permission(self, user, permission):
        if user not in self.user_permissions:
            self.user_permissions[user] = 0
        self.user_permissions[user] |= permission
    
    def revoke_permission(self, user, permission):
        if user in self.user_permissions:
            self.user_permissions[user] &= ~permission
    
    def check_permission(self, user, permission):
        if user not in self.user_permissions:
            return False
        return (self.user_permissions[user] & permission) != 0
    
    def get_user_permissions(self, user):
        if user not in self.user_permissions:
            return 0
        return self.user_permissions[user]

# 使用示例
ps = PermissionSystem()

# 授予权限
ps.grant_permission("alice", ps.READ | ps.WRITE)
ps.grant_permission("bob", ps.READ | ps.EXECUTE | ps.ADMIN)

# 检查权限
print(f"Alice有读权限: {ps.check_permission('alice', ps.READ)}")
print(f"Alice有管理权限: {ps.check_permission('alice', ps.ADMIN)}")
print(f"Bob有执行权限: {ps.check_permission('bob', ps.EXECUTE)}")

# 撤销权限
ps.revoke_permission("bob", ps.ADMIN)
print(f"Bob撤销后的管理权限: {ps.check_permission('bob', ps.ADMIN)}")
```

### 3.2 网络协议处理

```python
class IPHeader:
    def __init__(self, version=4, ihl=5, tos=0, total_length=0, 
                 identification=0, flags=0, fragment_offset=0,
                 ttl=64, protocol=6, checksum=0, source_ip=0, dest_ip=0):
        self.version = version
        self.ihl = ihl
        self.tos = tos
        self.total_length = total_length
        self.identification = identification
        self.flags = flags
        self.fragment_offset = fragment_offset
        self.ttl = ttl
        self.protocol = protocol
        self.checksum = checksum
        self.source_ip = source_ip
        self.dest_ip = dest_ip
    
    def pack(self):
        # 版本和首部长度（各4位）
        version_ihl = (self.version << 4) | self.ihl
        
        # 标志和片偏移（标志3位，偏移13位）
        flags_fragment = (self.flags << 13) | self.fragment_offset
        
        # 打包为32位整数列表
        return [
            (version_ihl << 16) | (self.tos << 8) | self.total_length,
            (self.identification << 16) | flags_fragment,
            (self.ttl << 24) | (self.protocol << 16) | self.checksum,
            self.source_ip,
            self.dest_ip
        ]
    
    @classmethod
    def unpack(cls, data):
        version_ihl = (data[0] >> 24) & 0xFF
        version = version_ihl >> 4
        ihl = version_ihl & 0x0F
        
        tos = (data[0] >> 16) & 0xFF
        total_length = data[0] & 0xFFFF
        
        identification = data[1] >> 16
        flags_fragment = data[1] & 0xFFFF
        flags = flags_fragment >> 13
        fragment_offset = flags_fragment & 0x1FFF
        
        ttl = (data[2] >> 24) & 0xFF
        protocol = (data[2] >> 16) & 0xFF
        checksum = data[2] & 0xFFFF
        
        source_ip = data[3]
        dest_ip = data[4]
        
        return cls(version, ihl, tos, total_length, identification,
                  flags, fragment_offset, ttl, protocol, checksum,
                  source_ip, dest_ip)

# 使用示例
header = IPHeader(version=4, ihl=5, total_length=60, protocol=6)
packed_data = header.pack()
print(f"打包后的数据: {[hex(x) for x in packed_data]}")

unpacked_header = IPHeader.unpack(packed_data)
print(f"解包后的版本: {unpacked_header.version}")
print(f"解包后的协议: {unpacked_header.protocol}")
```

### 3.3 图像处理

```python
class ImageProcessor:
    @staticmethod
    def extract_rgb_channels(pixel):
        """从32位像素值中提取RGB通道"""
        r = (pixel >> 16) & 0xFF
        g = (pixel >> 8) & 0xFF
        b = pixel & 0xFF
        return r, g, b
    
    @staticmethod
    def combine_rgb_channels(r, g, b):
        """将RGB通道组合为32位像素值"""
        return (r << 16) | (g << 8) | b
    
    @staticmethod
    def grayscale(pixel):
        """将像素转换为灰度"""
        r, g, b = ImageProcessor.extract_rgb_channels(pixel)
        # 使用加权平均
        gray = int(0.299 * r + 0.587 * g + 0.114 * b)
        return ImageProcessor.combine_rgb_channels(gray, gray, gray)
    
    @staticmethod
    def invert_colors(pixel):
        """反转颜色"""
        r, g, b = ImageProcessor.extract_rgb_channels(pixel)
        r = 255 - r
        g = 255 - g
        b = 255 - b
        return ImageProcessor.combine_rgb_channels(r, g, b)
    
    @staticmethod
    def adjust_brightness(pixel, factor):
        """调整亮度"""
        r, g, b = ImageProcessor.extract_rgb_channels(pixel)
        r = min(255, max(0, int(r * factor)))
        g = min(255, max(0, int(g * factor)))
        b = min(255, max(0, int(b * factor)))
        return ImageProcessor.combine_rgb_channels(r, g, b)

# 使用示例
pixel = 0xFF6B34  # 橙色
r, g, b = ImageProcessor.extract_rgb_channels(pixel)
print(f"原始像素 RGB: ({r}, {g}, {b})")

gray_pixel = ImageProcessor.grayscale(pixel)
gray_r, gray_g, gray_b = ImageProcessor.extract_rgb_channels(gray_pixel)
print(f"灰度像素 RGB: ({gray_r}, {gray_g}, {gray_b})")

inverted_pixel = ImageProcessor.invert_colors(pixel)
inv_r, inv_g, inv_b = ImageProcessor.extract_rgb_channels(inverted_pixel)
print(f"反转像素 RGB: ({inv_r}, {inv_g}, {inv_b})")

bright_pixel = ImageProcessor.adjust_brightness(pixel, 1.5)
bright_r, bright_g, bright_b = ImageProcessor.extract_rgb_channels(bright_pixel)
print(f"增亮像素 RGB: ({bright_r}, {bright_g}, {bright_b})")
```

### 3.4 压缩算法

```python
class SimpleCompressor:
    @staticmethod
    def compress_rle(data):
        """简单的游程编码压缩"""
        if not data:
            return b''
        
        compressed = []
        current_char = data[0]
        count = 1
        
        for char in data[1:]:
            if char == current_char and count < 255:
                count += 1
            else:
                # 将计数和字符打包为一个字节
                compressed.append((count << 8) | current_char)
                current_char = char
                count = 1
        
        compressed.append((count << 8) | current_char)
        return bytes(compressed)
    
    @staticmethod
    def decompress_rle(compressed_data):
        """解压游程编码数据"""
        decompressed = []
        
        for i in range(0, len(compressed_data), 2):
            if i + 1 < len(compressed_data):
                count = compressed_data[i]
                char = compressed_data[i + 1]
                decompressed.extend([char] * count)
        
        return bytes(decompressed)
    
    @staticmethod
    def pack_bits(data):
        """位打包压缩"""
        packed = []
        current_byte = 0
        bit_count = 0
        
        for byte in data:
            # 假设每个值只有低4位有效
            current_byte = (current_byte << 4) | (byte & 0x0F)
            bit_count += 4
            
            if bit_count == 8:
                packed.append(current_byte)
                current_byte = 0
                bit_count = 0
        
        if bit_count > 0:
            current_byte <<= (8 - bit_count)
            packed.append(current_byte)
        
        return bytes(packed)
    
    @staticmethod
    def unpack_bits(packed_data):
        """解压位打包数据"""
        unpacked = []
        
        for byte in packed_data:
            # 提取高4位和低4位
            high = (byte >> 4) & 0x0F
            low = byte & 0x0F
            unpacked.extend([high, low])
        
        return bytes(unpacked)

# 使用示例
data = b'AAAABBBCCCD'
compressed = SimpleCompressor.compress_rle(data)
print(f"原始数据: {data}")
print(f"压缩后: {compressed}")
decompressed = SimpleCompressor.decompress_rle(compressed)
print(f"解压后: {decompressed}")

# 位打包示例
data_to_pack = bytes([0x0A, 0x0B, 0x0C, 0x0D])
packed = SimpleCompressor.pack_bits(data_to_pack)
print(f"位打包: {data_to_pack.hex()} -> {packed.hex()}")
unpacked = SimpleCompressor.unpack_bits(packed)
print(f"位解包: {packed.hex()} -> {unpacked.hex()}")
```
