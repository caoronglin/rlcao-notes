# 文献阅读

```dataviewjs
const papers = dv.pages('"notes/bio"').where(p => p.citekey);
const totalPapers = papers.length;
const status = {
  toRead: papers.where(p => p.阅读状态 && p.阅读状态.contains("to_read")).length,
  reading: papers.where(p => p.阅读状态 && p.阅读状态.contains("reading")).length,
  done: papers.where(p => p.阅读状态 && (p.阅读状态.contains("read") || p.阅读状态.contains("reviewed"))).length,
};
const pct = totalPapers > 0 ? Math.round(status.done / totalPapers * 100) : 0;

let html = `
<div class="dash-read-card">
  <div class="dash-section-kicker">LITERATURE</div>
  <div class="dash-read-head">
    <div>
      <div class="dash-read-title">文献阅读进度</div>
      <div class="dash-read-sub">${status.done}/${totalPapers} 已完成</div>
    </div>
    <div class="dash-read-pct">${pct}%</div>
  </div>
  <div class="dash-progress"><div style="width:${pct}%"></div></div>
  <div class="dash-stat-grid three compact">
    <div class="dash-stat"><b>${status.toRead}</b><span>待读</span></div>
    <div class="dash-stat"><b>${status.reading}</b><span>在读</span></div>
    <div class="dash-stat"><b>${status.done}</b><span>已读</span></div>
  </div>
</div>`;
dv.span(html);

const reading = papers.where(p => p.阅读状态 && p.阅读状态.contains("reading"));
if (reading.length > 0) {
  dv.header(4, "正在阅读");
  dv.table(["文献", "作者"], reading.sort(p => p.日期, 'desc').limit(4).map(p => [p.file.link, p.作者 || "—"]));
}

dv.header(4, "最新添加");
dv.table(["文献", "日期"], papers.sort(p => p.日期, 'desc').limit(4).map(p => [p.file.link, p.日期 || "—"]));
```
