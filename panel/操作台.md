// 任务仍由 Dataview 写回原笔记；首页分组不改变项目状态或历史清单。
const today=window.moment().format("YYYY-MM-DD");
const projects=Array.from(dv.pages('"vault/项目"').where(p=>p.type==="project"));
const projectFor=p=>projects.filter(q=>q.file.path===p.file.path || (q.file.folder!=="vault/项目" && p.file.path.startsWith(q.file.folder+"/"))).sort((a,b)=>b.file.folder.length-a.file.folder.length)[0];
const groups={today:[],active:[],history:[]};
const seen=new Set();
const pages=dv.pages('"vault/项目"').concat(dv.pages('"vault/实验记录"')).concat(dv.pages('"vault/日记"').where(p=>p.file.name===today));
for(const p of pages){
  if(p.type==="guide") continue;
  const isDaily=p.file.path===`vault/日记/${today}.md`;
  const owner=projectFor(p);
  const linkedProject=p.project?.path?dv.page(p.project.path):null;
  const active=p.type==="experiment"?p.status==="进行中":(owner?.status==="进行中" || linkedProject?.status==="进行中");
  for(const t of p.file.tasks||[]){
    const key=t.path+":"+t.line;
    if(t.completed || ["-",">"] .includes(t.status) || !t.text.trim() || seen.has(key)) continue;
    if(isDaily && t.section?.subpath==="明日计划") continue;
    seen.add(key);
    groups[isDaily?"today":active?"active":"history"].push(t);
  }
}
const sortTasks=tasks=>tasks.sort((a,b)=>{
  const due=t=>t.due?.toISODate?.()||"9999-12-31";
  return due(a).localeCompare(due(b)) || a.path.localeCompare(b.path) || a.line-b.line;
});
const render=async(tasks,container)=>{if(tasks.length)await dv.api.taskList(sortTasks(tasks),true,container,dv.component,dv.currentFilePath);};
const daily=dv.el("section","",{cls:"wf-task-section"});
daily.createEl("h3",{text:`今日计划 · ${groups.today.length}`});
if(groups.today.length) await render(groups.today,daily);
else daily.createEl("p",{text:"今天还没有填写计划。点“今日日记”，写下最想完成的三件事。",cls:"wf-empty"});
const current=dv.el("section","",{cls:"wf-task-section"});
current.createEl("h3",{text:`进行中项目与实验 · ${groups.active.length}`});
if(groups.active.length) await render(groups.active,current);
else current.createEl("p",{text:"进行中的记录暂无未完成清单。可在项目笔记中补充下一步。",cls:"wf-empty"});
if(groups.history.length){
  const history=dv.el("details","",{cls:"wf-history"});
  history.createEl("summary",{text:`历史与待整理清单 · ${groups.history.length} 项`});
  history.createEl("p",{text:"来自待整理、未标状态或已完成的项目与实验；保留原记录，不计入今日计划。",cls:"wf-empty"});
  await render(groups.history,history);
}
