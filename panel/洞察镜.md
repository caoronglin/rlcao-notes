# 洞察镜

```dataviewjs
// dataview 的 mtime 是 Luxon 对象，moment() 直接接收会退化成“当前时间”
const M = t => (t && typeof t.toJSDate === "function") ? moment(t.toJSDate()) : moment(t);
const now = moment();
const SKIP = /^(panel|muban)\/|^Home\.md$/;
const ageOf = p => now.diff(M(p.file.mtime), "days");

const pages = dv.pages()
  .where(p => !SKIP.test(p.file.path))
  .where(p => !p.file.path.startsWith("vault/日记"));
const recent = pages.where(p => ageOf(p) <= 30);

// ── 1. 被遗忘的内容：优先 60 天，样本不足时自动放宽到 30 天 ──
const stale = d => pages.where(p => ageOf(p) > d).sort(p => p.file.mtime, "asc").limit(5);
let forgotten = stale(60);
let forgottenLabel = "超过两个月";
if (!forgotten.length) {
  forgotten = stale(30);
  forgottenLabel = "超过一个月";
}

// ── 2. 近期热点：近 30 天改动笔记的标签统计 ──
const tagCount = {};
recent.forEach(p => {
  (p.file.tags || []).forEach(t => {
    const k = String(t).replace(/^#/, "");
    if (k) tagCount[k] = (tagCount[k] || 0) + 1;
  });
});
const hotTags = Object.entries(tagCount).sort((a, b) => b[1] - a[1]).slice(0, 8);

// ── 3. 注意力分布：近 30 天各顶层目录的改动量 ──
const bucket = {};
recent.forEach(p => {
  const top = (p.file.path.split("/")[0]) || "根目录";
  bucket[top] = (bucket[top] || 0) + 1;
});
const bars = Object.entries(bucket).sort((a, b) => b[1] - a[1]).slice(0, 5);
const maxBar = bars.length ? bars[0][1] : 1;

// ── 4. 行动建议 ──
const suggestions = [];
if (forgotten.length) suggestions.push(`回顾《${forgotten[0].file.name}》，已 ${ageOf(forgotten[0])} 天未动`);
if (hotTags.length) suggestions.push(`近期注意力集中在「${hotTags[0][0]}」，确认是否要收敛或产出`);
if (bars.length < 3) suggestions.push("注意力过度集中，考虑给冷门目录分配时间");
if (!suggestions.length) suggestions.push("节奏均衡，保持当前状态");

const forgottenHtml = forgotten.length
  ? forgotten.map(p => `<div class="kh-item">
      <span class="kh-dot muted"></span>
      <span class="kh-item-main">
        <a class="internal-link kh-item-title" data-href="${p.file.path}">${p.file.name}</a>
        <span class="kh-item-meta">${ageOf(p)} 天未动</span>
      </span>
    </div>`).join("")
  : `<div class="kh-empty">没有长期搁置的笔记</div>`;

const tagsHtml = hotTags.length
  ? `<div class="kh-tag-row">${hotTags.map(([t, c]) => `<a class="kh-tag internal-link" data-href="${t}">#${t} ${c}</a>`).join("")}</div>`
  : `<div class="kh-empty">近 30 天没有打标签</div>`;

const barsHtml = bars.length
  ? `<div class="kh-bars">${bars.map(([name, n]) => `<div class="kh-bar-row">
      <span class="kh-bar-label">${name}</span>
      <span class="kh-bar-track"><span class="kh-bar-fill" style="width:${Math.round(n / maxBar * 100)}%"></span></span>
      <span class="kh-bar-num">${n}</span>
    </div>`).join("")}</div>`
  : `<div class="kh-empty">近 30 天没有改动</div>`;

const insight = (title, text, extra = "") => `
<div class="kh-insight-card">
  <div class="kh-insight-title">${title}</div>
  <div class="kh-insight-text">${text}</div>
  ${extra}
</div>`;

dv.span(`
<div class="kh-insight">
  ${insight("🕰 被遗忘的内容", `这些笔记已${forgottenLabel}没有回看，可能是被埋掉的想法。`, forgottenHtml)}
  ${insight("🔥 近期热点", "近 30 天你在这些标签上最活跃。", tagsHtml)}
  ${insight("🎯 注意力分布", "近 30 天各目录的改动量，看看有没有偏科。", barsHtml)}
  ${insight("💡 行动建议", suggestions.map(s => `· ${s}`).join("<br>"))}
</div>
`);
```
