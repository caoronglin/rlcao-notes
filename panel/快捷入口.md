    ["实验管理", "vault/工作台/实验管理.md"],

];

const grid = dv.container.createDiv({ cls: "wf-entry-grid" });
const accents = ["var(--ant-primary)", "var(--ant-success)", "var(--ant-purple)"];
groups.forEach(([label, entries], i) => {
  const group = grid.createDiv({ cls: "wf-entry-group" });
  group.style.setProperty("--wf-accent", accents[i % accents.length]);
  group.createEl("h3", { text: label });
  for (const [text, path] of entries) {
    if (app.vault.getAbstractFileByPath(path)) {
      const a = group.createEl("a", { text, href: path, cls: "wf-entry-link internal-link" });
      a.setAttr("data-href", path);
    } else {
      group.createEl("span", { text: `${text}（入口待修复）`, cls: "wf-empty" });
    }
  }
});
