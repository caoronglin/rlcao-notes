```dataviewjs
// 今日计划 + 进行中项目/实验 + 历史折叠。勾选仍写回原笔记，首页不判断项目是否完成。
const today = window.moment().format("YYYY-MM-DD");
const projects = Array.from(dv.pages('"vault/项目"').where(p => p.type === "project"));
const projectFor = p => projects.filter(q =>
  q.file.path === p.file.path ||
  (q.file.folder !== "vault/项目" && p.file.path.startsWith(q.file.folder + "/"))
).sort((a, b) => b.file.folder.length - a.file.folder.length)[0];

const groups = { today: [], active: [], history: [] };
const seen = new Set();
const dailyPage = dv.page(`vault/日记/${today}`);
const pages = dv.pages('"vault/项目"')
  .concat(dv.pages('"vault/实验记录"'))
  .concat(dv.pages('"vault/日记"').where(p => p.file.name === today));

for (const p of pages) {
  if (p.type === "guide") continue;
  const isDaily = p.file.path === `vault/日记/${today}.md`;
  const owner = projectFor(p);
  const linkedProject = p.project?.path ? dv.page(p.project.path) : null;
  const active = p.type === "experiment"
    ? p.status === "进行中"
    : (owner?.status === "进行中" || linkedProject?.status === "进行中");
  for (const t of p.file.tasks || []) {
    const key = t.path + ":" + t.line;
    if (t.completed || ["-", ">"].includes(t.status) || !t.text.trim() || seen.has(key)) continue;
    if (isDaily && t.section?.subpath === "明日计划") continue;
    seen.add(key);
    groups[isDaily ? "today" : active ? "active" : "history"].push(t);
  }
}

const sortTasks = tasks => tasks.sort((a, b) => {
  const due = t => t.due?.toISODate?.() || "9999-12-31";
  return due(a).localeCompare(due(b)) || a.path.localeCompare(b.path) || a.line - b.line;
});
const render = async (tasks, container) => {
  if (tasks.length) await dv.api.taskList(sortTasks(tasks), true, container, dv.component, dv.currentFilePath);
};

// ---- 左：今日计划（带完成度）----
const root = dv.container.createDiv({ cls: "wf-root" });
const cols = root.createDiv({ cls: "wf-cols" });

const todayCard = cols.createDiv({ cls: "wf-card" });
todayCard.style.setProperty("--wf-accent", "var(--ant-primary)");
const todayTitle = todayCard.createDiv({ cls: "wf-section-title", text: "今日计划" });
todayTitle.createSpan({ cls: "wf-badge", text: `${groups.today.length} 项待办` });

const allToday = (dailyPage?.file.tasks || []).filter(t =>
  t.text.trim() && t.section?.subpath !== "明日计划"
);
const doneToday = allToday.filter(t => t.completed).length;
if (allToday.length) {
  const ratio = Math.round((doneToday / allToday.length) * 100);
  todayCard.createDiv({ cls: "wf-progress is-active" }).createEl("span").style.width = ratio + "%";
  todayCard.createDiv({ cls: "wf-progress-text" }).createEl("span", {
    text: `已完成 ${doneToday} / ${allToday.length}`
  });
}
if (groups.today.length) await render(groups.today, todayCard);
else todayCard.createEl("p", { cls: "wf-empty", text: "今天还没有未完成的计划。点「今日日记」写下最想完成的三件事。" });

// ---- 右：进行中项目与实验 ----
const activeCard = cols.createDiv({ cls: "wf-card" });
activeCard.style.setProperty("--wf-accent", "var(--ant-success)");
const activeTitle = activeCard.createDiv({ cls: "wf-section-title", text: "进行中项目与实验" });
activeTitle.createSpan({ cls: "wf-badge", text: `${groups.active.length} 项待办` });
if (groups.active.length) await render(groups.active, activeCard);
else activeCard.createEl("p", { cls: "wf-empty", text: "进行中的项目/实验暂无未完成清单，可在对应笔记里补一条下一步。" });

// ---- 下：历史与待整理 ----
if (groups.history.length) {
  const files = new Set(groups.history.map(t => t.path)).size;
  const card = root.createDiv({ cls: "wf-card" });
  card.style.setProperty("--wf-accent", "var(--ant-text-3)");
  const details = card.createEl("details", { cls: "wf-history" });
  details.createEl("summary", { text: `历史与待整理清单 · ${groups.history.length} 项 / ${files} 篇笔记` });
  details.createEl("p", {
    cls: "wf-empty",
    text: "来自未标状态或已归档的项目与实验，原记录不动，不计入今日计划。"
  });
  await render(groups.history, details);
}
```
