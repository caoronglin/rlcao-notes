## 7.1 结构体

### 结构体概述

- **构造数据类型**：将不同类型的数据组合成整体
- **特点**：

- 成员可以是基本类型或构造类型
- `struct`是关键字，不可省略
- 类型定义不分配内存

### 结构体定义

```c
struct [结构体名] {
    类型1 成员1;
    类型2 成员2;
    // ...
};
```

**示例**：

```c
struct student {
    int num;
    char name[20];
    char gender;
    int age;
    float score;
};
```

### 结构体变量定义

| 方式 | 示例代码 |
| --- | --- |
| 先定义类型后定义变量 | `struct student stu1, stu2;` |
| 定义类型同时定义变量 | `struct student { /*成员*/ } stu1, stu2;` |
| 直接定义变量 | `struct { /*成员*/ } stu1, stu2;`（无名结构体，只能使用一次） |

### 结构体变量引用

- 只能引用成员，不能整体引用
- 成员运算符`.`优先级1，左结合性

```c
stu1.num = 101;
strcpy(stu1.name, "Zhang San");
```

### 结构体初始化

```c
// 方式1：先定义后初始化
struct student stu1 = {101, "Li Ming", 'M', 20, 90.5};

// 方式2：定义类型时初始化
struct student {
    // 成员
} stu1 = {102, "Wang Fang", 'F', 19, 88.0};

// 方式3：无名结构体初始化
struct {
    // 成员
} stu1 = {103, "Zhao Wei", 'M', 21, 92.5};
```

### 结构体嵌套

```c
struct date {
    int year;
    int month;
    int day;
};

struct student {
    char no[10];
    char name[20];
    struct date birthday; // 嵌套结构体
    float score;
};
```

### 结构体数组

```c
// 定义与初始化
struct student stu[3] = {
    {"S001", "Zhang", {2000,9,1}, 85.5},
    {"S002", "Wang", {2001,3,12}, 92.0},
    {"S003", "Li", {1999,11,20}, 78.5}
};

// 访问元素
printf("%s's score: %.1f", stu[1].name, stu[1].score);
```

### 结构体指针

```c
struct student stu1, *p = &stu1;

// 三种访问方式
stu1.num = 101;
(*p).num = 102; 
p->num = 103; // 最常用
```

### 结构体与函数

#### 1. 结构体作为函数参数

```c
// 值传递
void printStudent(struct student s) {
    printf("Name: %s\nScore: %.1f", s.name, s.score);
}

// 指针传递
void modifyScore(struct student *p, float newScore) {
    p->score = newScore;
}
```

#### 2. 结构体作为返回值

```c
// 返回结构体
struct student createStudent() {
    struct student s;
    // 初始化s
    return s;
}

// 返回结构体指针
struct student* findStudent(int id) {
    // 查找逻辑
    return &stu[id];
}
```

## 7.2 共用体

### 共用体概述

- **联合体**：不同类型数据共享同一内存空间
- **特点**：

- 内存大小 = 最大成员的大小
- 任一时刻只有一个成员有效
- `union`为关键字

### 共用体定义与使用

```c
union data {
    int i;
    char ch;
    float f;
};

union data a;
a.i = 10;       // 存入整数
a.f = 220.5;    // 覆盖原有数据，存入浮点数
```

### 结构体嵌套共用体

```c
struct person {
    int num;
    char name[15];
    char gender;
    char job;
    union {
        int classid;    // 学生班级
        char title[10]; // 教师职称
    } category;
};

// 使用示例
struct person p1;
p1.job = 's'; // 学生
p1.category.classid = 202;

struct person p2;
p2.job = 't'; // 教师
strcpy(p2.category.title, "Professor");
```

## 7.3 枚举类型

### 枚举类型定义

```c
enum 枚举类型名 { 枚举元素表 };
```

**示例**：

```c
enum week { sun, mon, tue, wed, thu, fri, sat };
enum color { RED, GREEN=5, BLUE };
```

### 枚举使用规则

1. 元素值默认从0开始递增
2. 可显式赋值，后续元素自动+1
3. 枚举变量只能赋枚举元素值
4. 允许与整数进行运算

```c
enum week today = mon;
today = (enum week)2; // 强制转换

if (today == tue)
    printf("Tuesday");
```

## 7.4 用typedef定义类型

### typedef作用

为已有类型定义新名称，增强可读性

### 使用方式

```c
typedef 原类型 新类型名;
```

**示例**：

```c
typedef int INTEGER;
typedef float REAL;
typedef struct {
    int month;
    int day;
    int year;
} DATE;

// 使用新类型
INTEGER count = 10;
REAL price = 99.99;
DATE birthday = {12, 15, 1990};
```

### 复杂类型定义

```c
// 数组类型
typedef int ARRAY[10];
ARRAY a, b; // 等价于 int a[10], b[10];

// 指针类型
typedef char* STRING;
STRING p, arr[5]; // char *p, *arr[5]

// 结构体指针
typedef struct student* PSTUDENT;
PSTUDENT p1, p2;
```

## 综合应用示例

### 结构体数组排序

```c
#include <stdio.h>
#include <string.h>

typedef struct student {
    char name[15];
    int age;
    float score;
} STUDENT;

void sortStudents(STUDENT arr[], int n) {
    for(int i=0; i<n-1; i++) {
        for(int j=i+1; j<n; j++) {
            // 按成绩降序，成绩相同按年龄升序
            if(arr[i].score < arr[j].score || 
              (arr[i].score == arr[j].score && arr[i].age > arr[j].age)) {
                STUDENT temp = arr[i];
                arr[i] = arr[j];
                arr[j] = temp;
            }
        }
    }
}

int main() {
    STUDENT stu[5] = {
        {"Alice", 20, 88.5},
        {"Bob", 19, 92.0},
        {"Charlie", 21, 88.5},
        {"David", 20, 76.5},
        {"Eva", 22, 95.0}
    };
    
    sortStudents(stu, 5);
    
    for(int i=0; i<5; i++) {
        printf("%-10s %2d %.1f\n", stu[i].name, stu[i].age, stu[i].score);
    }
    return 0;
}
```

### 枚举类型应用

```c
#include <stdio.h>

enum BOOL { FALSE, TRUE };

enum BOOL isPrime(int n) {
    if(n < 2) return FALSE;
    for(int i=2; i*i<=n; i++) {
        if(n % i == 0) return FALSE;
    }
    return TRUE;
}

int main() {
    int num;
    printf("Enter a number: ");
    scanf("%d", &num);
    
    if(isPrime(num) == TRUE) {
        printf("%d is prime\n", num);
    } else {
        printf("%d is not prime\n", num);
    }
    return 0;
}
```
