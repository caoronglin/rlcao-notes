# 操作台

```dataviewjs
// dataview 的 mtime/ctime/due 是 Luxon 对象，moment() 直接接收会退化成“当前时间”
const M = t => (t && typeof t.toJSDate === "function") ? moment(t.toJSDate()) : moment(t);

// ── 今日行动：今天真正要推进的事情 ──
const actions = [];
dv.pages('"vault/项目"').forEach(p => {
  p.file.tasks.where(t => !t.completed).forEach(t => {
    actions.push({ text: t.text, src: p.file, due: t.due });
  });
});
const now = moment();
const todayStr = now.format("YYYY-MM-DD");
const todayDiary = dv.page(`vault/日记/${todayStr}`);
if (todayDiary) {
  todayDiary.file.tasks.where(t => !t.completed).forEach(t => {
    actions.push({ text: t.text, src: todayDiary.file, due: t.due });
  });
}
// 有截止日期的排前面
actions.sort((a, b) => {
  if (a.due && b.due) return M(a.due).diff(M(b.due));
  if (a.due) return -1;
  if (b.due) return 1;
  return 0;
});

// ── 今日输入：今天新进来的笔记 ──
const SKIP = /^(panel|muban)\/|^Home\.md$/;
const inScope = p => !SKIP.test(p.file.path) && !p.file.path.startsWith("vault/日记");
const createdToday = p => M(p.file.ctime).format("YYYY-MM-DD") === todayStr;

const inputs = dv.pages()
  .where(inScope).where(createdToday)
  .sort(p => p.file.ctime, "desc")
  .limit(6);

// ── 今日输出：今天改动过的既有笔记 ──
const outputs = dv.pages()
  .where(inScope)
  .where(p => M(p.file.mtime).format("YYYY-MM-DD") === todayStr && !createdToday(p))
  .sort(p => p.file.mtime, "desc")
  .limit(6);

// 链接渲染：HTML 里的 markdown 链接不会被自动解析，必须用 a[data-href]
const fileLink = f => `<a class="internal-link kh-item-title" data-href="${f.path}">${f.name}</a>`;
const renderList = (rows, render) => rows.length
  ? rows.map(render).join("")
  : `<div class="kh-empty">还没有记录</div>`;

const actionHtml = renderList(actions.slice(0, 6), t => {
  const overdue = t.due && M(t.due).isBefore(now, "day");
  const dueTxt = t.due ? ` · 截止 ${M(t.due).format("M月D日")}` : "";
  return `<div class="kh-item">
    <span class="kh-dot ${overdue ? "warn" : ""}"></span>
    <span class="kh-item-main">
      <span class="kh-item-title">${t.text}</span>
      <span class="kh-item-meta">${fileLink(t.src)}${dueTxt}</span>
    </span>
  </div>`;
});

const inputHtml = renderList(inputs, p => `<div class="kh-item">
  <span class="kh-dot"></span>
  <span class="kh-item-main">
    ${fileLink(p.file)}
    <span class="kh-item-meta">${p.file.folder || "库根"}</span>
  </span>
</div>`);

const outputHtml = renderList(outputs, p => `<div class="kh-item">
  <span class="kh-dot"></span>
  <span class="kh-item-main">
    ${fileLink(p.file)}
    <span class="kh-item-meta">${M(p.file.mtime).format("HH:mm")} 更新 · ${p.file.folder || "库根"}</span>
  </span>
</div>`);

const card = (icon, title, count, body) => `
<div class="kh-card">
  <div class="kh-card-head">
    <div class="kh-card-icon">${icon}</div>
    <div class="kh-card-title">${title}</div>
    <div class="kh-card-count">${count}</div>
  </div>
  <div class="kh-card-body">${body}</div>
</div>`;

dv.span(`
<div class="kh-board">
  ${card("🎯", "今日行动", actions.length, actionHtml)}
  ${card("📥", "今日输入", inputs.length, inputHtml)}
  ${card("📤", "今日输出", outputs.length, outputHtml)}
</div>
`);
```
