# 项目看板

```dataviewjs
const projects = dv.pages('"项目"').where(p => p.finished != true);

if (projects.length === 0) {
  dv.span(`
<div class="dash-empty-state">
  <div class="dash-empty-icon">✨</div>
  <div class="dash-empty-title">暂无进行中的项目</div>
  <div class="dash-empty-sub">用 Templater 模板新建一个项目开始吧</div>
</div>`);
} else {
  let html = `<div class="dash-list">`;
  for (const p of projects.sort(p => p.file.ctime, 'desc').limit(6)) {
    const priority = p.priority || "normal";
    const badge = priority == "high" ? "urgent" : priority == "medium" ? "medium" : "normal";
    html += `
<div class="dash-list-card ${badge}">
  <div class="dash-list-head">
    <div class="dash-list-title">${p.file.link}</div>
    <div class="dash-badge ${badge}">${priority}</div>
  </div>
  ${p.nextAction ? `<div class="dash-list-sub">${p.nextAction}</div>` : ""}
  <div class="dash-list-meta">创建 ${moment(p.file.ctime).format("M月D日")}${p.created ? ` · 启动 ${p.created}` : ""}</div>
</div>`;
  }
  html += `</div>`;
  dv.span(html);
}
```
