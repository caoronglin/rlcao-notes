```dataviewjs
const now = moment();
const hw = dv.page('"项目/细胞生物学作业"');
const hwDue = hw ? moment(hw.due) : null;
const hwDays = hwDue ? hwDue.diff(now, 'days') : null;
const newYear = moment("2027-01-01");
const nyDays = newYear.diff(now, 'days');

const items = [];
if (hwDue && hwDays >= 0) {
  items.push({ label: '细胞生物学作业', days: hwDays, date: hwDue.format('M月D日'), type: hwDays <= 3 ? 'urgent' : hwDays <= 7 ? 'medium' : 'normal' });
}
items.push({ label: '2027 年元旦', days: nyDays, date: '1月1日', type: 'normal' });

let html = `<div class="dash-countdown-grid">`;
for (const it of items) {
  html += `
<div class="dash-countdown-card ${it.type}">
  <div class="dash-countdown-top">
    <span>${it.label}</span>
    <em>${it.date}</em>
  </div>
  <div class="dash-countdown-num">${it.days}</div>
  <div class="dash-countdown-unit">days left</div>
</div>`;
}
html += `</div>`;
dv.span(html);
```
