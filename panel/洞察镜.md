const projects=dv.pages('"vault/项目"').where(p=>p.type==="project").sort(p=>p.file.mtime,"desc").limit(6);
dv.header(3,"项目进展");

const experiments=dv.pages('"vault/实验记录"').where(p=>p.type==="experiment").sort(p=>p.date||p.file.mtime,"desc").limit(5);
dv.header(3,"最近实验");
if(experiments.length)dv.table(["实验","日期","状态","项目"],experiments.map(p=>[p.file.link,p.date||"—",p.status||"未填写",p.project||"—"]));else dv.paragraph("还没有实验记录。从上方“新建实验”开始；方法资料保存在方法区。");
dv.header(3,"最近更新");
const recent=dv.pages('"vault"').where(p=>!p.file.path.startsWith("vault/工作台/")&&!p.file.name.endsWith("使用说明")&&p.type!=="guide").sort(p=>p.file.mtime,"desc").limit(5);
dv.list(recent.map(p=>p.file.link));
