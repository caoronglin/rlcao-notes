# 📅 本月日记

```dataviewjs
const now = moment();
const y = now.year();
const m = now.month() + 1;
const prefix = `${y}-${String(m).padStart(2, '0')}`;

const pages = dv.pages('"日记"')
  .where(p => p.file.name.startsWith(prefix))
  .sort(p => p.file.name, 'asc');

const daysInMonth = now.daysInMonth();
const diarySet = new Set(pages.map(p => p.file.name));
const diaryCount = pages.length;
const pct = Math.round(diaryCount / daysInMonth * 100);

let html = `<div class="dash-mini-split">`;

html += `<div class="dash-calendar-box">`;
html += `<div class="dash-mini-title">${m}月记录</div>`;
html += `<div class="dash-mini-sub">${diaryCount}/${daysInMonth} 天 · ${pct}%</div>`;
html += `<div class="dash-calendar-grid">`;

for (const h of ['一','二','三','四','五','六','日']) {
  html += `<div class="dash-calendar-head">${h}</div>`;
}

const firstDay = moment(`${prefix}-01`, "YYYY-MM-DD").day();
const pad = (firstDay + 6) % 7;
for (let i = 0; i < pad; i++) html += `<div></div>`;

for (let d = 1; d <= daysInMonth; d++) {
  const dateStr = `${prefix}-${String(d).padStart(2, '0')}`;
  const hasDiary = diarySet.has(dateStr);
  const isToday = dateStr === now.format("YYYY-MM-DD");
  html += `<div class="dash-calendar-day ${hasDiary ? 'has-diary' : ''} ${isToday ? 'is-today' : ''}" title="${dateStr}">${d}</div>`;
}
html += `</div></div>`;

html += `<div class="dash-recent-box">`;
html += `<div class="dash-mini-title">最近日记</div>`;
const recent = pages.sort(p => p.file.name, 'desc').slice(0, 5);
for (const p of recent) {
  html += `<div class="dash-line-item">📄 ${p.file.link}</div>`;
}
if (recent.length === 0) html += `<div class="dash-empty">本月暂无日记</div>`;
html += `</div></div>`;

dv.span(html);
```
