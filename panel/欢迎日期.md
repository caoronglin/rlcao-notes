```dataviewjs
const now = moment();
const week = ["日","一","二","三","四","五","六"];
const today = now.format("YYYY年M月D日");
const weekDay = week[now.day()];
const weekNum = now.isoWeek();

const allPages = dv.pages().length;
const notePages = dv.pages('"notes"').length;
const blogPages = dv.pages('"_posts"').length;
const projects = dv.pages('"项目"').where(p => p.finished != true).length;
const homework = dv.pages('"项目"').where(p => p.course == "细胞生物学" && p.finished != true).length;
const papers = dv.pages('"notes/bio"').where(p => p.citekey).length;

const cards = [
  { icon: "📄", label: "笔记", count: notePages, sub: "notes" },
  { icon: "📝", label: "博客", count: blogPages, sub: "_posts" },
  { icon: "🔬", label: "文献", count: papers, sub: "bio" },
  { icon: "📋", label: "项目", count: projects, sub: "active" },
  { icon: "⏰", label: "作业", count: homework, sub: "cell" },
];

let html = `
<div class="dash-welcome">
  <div class="dash-welcome-main">
    <div class="dash-eyebrow">OBSIDIAN WORKSPACE</div>
    <div class="dash-welcome-title">${today}</div>
    <div class="dash-welcome-sub">星期${weekDay} · 第 ${weekNum} 周 · 知识库共 ${allPages} 篇笔记</div>
  </div>
  <div class="dash-metric-row">`;

for (const c of cards) {
  html += `
    <div class="dash-metric-card">
      <div class="dash-metric-icon">${c.icon}</div>
      <div>
        <div class="dash-metric-num">${c.count}</div>
        <div class="dash-metric-label">${c.label} · ${c.sub}</div>
      </div>
    </div>`;
}

html += `</div></div>`;
dv.span(html);
```
