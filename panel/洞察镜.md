const dateText=value=>value?.toISODate?.()||value||"—";
const projects=dv.pages('"vault/项目"').where(p=>p.type==="project").sort(p=>p.file.mtime,"desc");
const active=projects.where(p=>p.status==="进行中");
dv.header(3,"正在推进");
if(active.length)dv.table(["项目","状态","截止日期"],active.map(p=>[p.file.link,p.status,dateText(p.due)]));
else dv.paragraph("暂无标记为“进行中”的项目。可在项目管理中按实际进度维护状态。");
const pending=projects.where(p=>!["进行中","完成","已完成","归档","已归档"].includes(String(p.status||"")));
if(pending.length){
  const details=dv.el("details","",{cls:"wf-history"});
  details.createEl("summary",{text:`待确认的项目状态 · ${pending.length} 个`});
  await dv.api.table(["项目","当前状态"],pending.map(p=>[p.file.link,p.status||"未填写"]),details,dv.component,dv.currentFilePath);
}
dv.header(3,"最近实验");
const experiments=dv.pages('"vault/实验记录"').where(p=>p.type==="experiment").sort(p=>p.date||p.file.mtime,"desc").limit(4);
if(experiments.length)dv.table(["实验","日期","状态"],experiments.map(p=>[p.file.link,dateText(p.date),p.status||"未填写"]));
else dv.paragraph("还没有实验记录。从上方“新建实验”开始，方法资料保存在方法区。");
dv.header(3,"接着上次的记录");
const recent=dv.pages('"vault"').where(p=>!p.file.path.startsWith("vault/工作台/") && !p.file.name.endsWith("使用说明") && p.type!=="guide").sort(p=>p.file.mtime,"desc").limit(5);
if(recent.length)dv.list(recent.map(p=>p.file.link));else dv.paragraph("这里会显示最近编辑的学习与研究记录。");
