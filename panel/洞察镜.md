// 项目 / 实验 / 最近记录只做只读汇总，状态以各笔记 frontmatter 为准
const dateText = v => v?.toISODate?.() || v || "—";
const projects = dv.pages('"vault/项目"').where(p => p.type === "project").sort(p => p.file.mtime, "desc");
const active = projects.where(p => p.status === "进行中");

const root = dv.container.createDiv({ cls: "wf-root" });
const cols = root.createDiv({ cls: "wf-cols" });

// ---- 正在推进 ----
const pCard = cols.createDiv({ cls: "wf-card" });
pCard.style.setProperty("--wf-accent", "var(--ant-primary)");
pCard.createDiv({ cls: "wf-section-title", text: "正在推进" })
  .createSpan({ cls: "wf-badge", text: `${active.length} 个项目` });
if (active.length) {
  await dv.api.table(
    ["项目", "状态", "截止日期"],
    active.map(p => [p.file.link, p.status, dateText(p.due)]),
    pCard, dv.component, dv.currentFilePath
  );
} else {
  pCard.createEl("p", { cls: "wf-empty", text: "没有标记为「进行中」的项目。在项目笔记 frontmatter 里把 status 改成进行中即可出现在这里。" });
}

// ---- 最近实验 ----
const eCard = cols.createDiv({ cls: "wf-card" });
eCard.style.setProperty("--wf-accent", "var(--ant-success)");
const experiments = dv.pages('"vault/实验记录"')
  .where(p => p.type === "experiment")
  .sort(p => p.date || p.file.mtime, "desc")
  .limit(5);
eCard.createDiv({ cls: "wf-section-title", text: "最近实验" })
  .createSpan({ cls: "wf-badge", text: `${experiments.length} 条` });
if (experiments.length) {
  await dv.api.table(
    ["实验", "日期", "状态"],
    experiments.map(p => [p.file.link, dateText(p.date), p.status || "未填写"]),
    eCard, dv.component, dv.currentFilePath
  );
} else {
  eCard.createEl("p", { cls: "wf-empty", text: "还没有实验记录。点首页「新建实验」开始，方法沉淀到 vault/方法。" });
}

// ---- 接着上次的记录 ----
const rCard = root.createDiv({ cls: "wf-card" });
rCard.style.setProperty("--wf-accent", "var(--ant-purple)");
const recent = dv.pages('"vault"')
  .where(p => !p.file.path.startsWith("vault/工作台/") && p.type !== "guide")
  .sort(p => p.file.mtime, "desc")
  .limit(6);
rCard.createDiv({ cls: "wf-section-title", text: "接着上次的记录" })
  .createSpan({ cls: "wf-badge", text: "按修改时间" });
if (recent.length) {
  const list = rCard.createEl("ul", { cls: "wf-recent" });
  for (const p of recent) {
    const li = list.createEl("li");
    li.createEl("a", { cls: "internal-link", text: p.file.name, href: p.file.path })
      .setAttr("data-href", p.file.path);
    li.createSpan({ cls: "wf-recent-time", text: window.moment(p.file.mtime.toMillis()).fromNow() });
  }
} else {
  rCard.createEl("p", { cls: "wf-empty", text: "暂无记录。" });
}

// ---- 待确认状态：折叠，不打扰 ----
const pending = projects.where(p => !["进行中", "完成", "已完成", "归档", "已归档"].includes(String(p.status || "")));
if (pending.length) {
  const p2 = root.createDiv({ cls: "wf-card" });
  p2.style.setProperty("--wf-accent", "var(--ant-warning)");
  const details = p2.createEl("details", { cls: "wf-history" });
  details.createEl("summary", { text: `待确认状态的项目 · ${pending.length} 个` });
  await dv.api.table(
    ["项目", "当前状态", "最近更新"],
    pending.map(p => [p.file.link, p.status || "未填写", dateText(p.file.mtime)]),
    details, dv.component, dv.currentFilePath
  );
}
