// 展示真实未完成任务；Dataview 原生任务列表支持直接勾选并写回原笔记。
const today=window.moment().format("YYYY-MM-DD");
const pages=dv.pages('"vault/项目"').concat(dv.pages('"vault/实验记录"')).concat(dv.pages('"vault/日记"').where(p=>p.file.name===today));
const seen=new Set();const tasks=[];
for(const p of pages){for(const t of p.file.tasks||[]){const key=t.path+":"+t.line;if(!t.completed&&t.text.trim()&&!seen.has(key)){seen.add(key);tasks.push(t);}}}
tasks.sort((a,b)=>String(a.due||"9999").localeCompare(String(b.due||"9999")));
if(tasks.length){dv.paragraph(`共 ${tasks.length} 项待办，优先显示前 12 项。`);dv.taskList(tasks.slice(0,12),false);}else{dv.paragraph("还没有待办。可以从今日日记或项目的“下一步”开始记录。");}
